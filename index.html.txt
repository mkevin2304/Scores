  </footer>

  <script>
    const SPORTS = {
      soccer: [
        { name: "Premier League", path: "soccer/eng.1" },
        { name: "La Liga", path: "soccer/esp.1" },
        { name: "Serie A", path: "soccer/ita.1" },
        { name: "Bundesliga", path: "soccer/ger.1" },
        { name: "Champions League", path: "soccer/uefa.champions" },
        { name: "All Soccer", path: "soccer/all" }
      ],
      basketball: [
        { name: "NBA", path: "basketball/nba" },
        { name: "NCAA Men's", path: "basketball/mens-college-basketball" }
      ],
      football: [
        { name: "NFL", path: "football/nfl" },
        { name: "College Football", path: "football/college-football" }
      ],
      baseball: [
        { name: "MLB", path: "baseball/mlb" }
      ],
      hockey: [
        { name: "NHL", path: "hockey/nhl" }
      ]
    };

    let currentSport = "soccer";
    let refreshInterval;

    async function fetchScoreboard(path) {
      const url = `https://site.api.espn.com/apis/site/v2/sports/${path}/scoreboard`;
      const res = await fetch(url);
      if (!res.ok) throw new Error("Failed to fetch");
      return res.json();
    }

    function renderMatch(event) {
      const comp = event.competitions[0];
      const home = comp.competitors.find(c => c.homeAway === "home");
      const away = comp.competitors.find(c => c.homeAway === "away");
      const status = comp.status.type;
      const isLive = status.state === "in";
      const isFinal = status.completed;

      return `
        <div class="match">
          <div class="team away">
            ${away.team.logo ? `<img class="logo" src="${away.team.logo}" alt="">` : ""}
            <span>${away.team.shortDisplayName || away.team.displayName}</span>
          </div>
          <div class="score">
            ${away.score || "-"} – ${home.score || "-"}
          </div>
          <div class="team home">
            <span>${home.team.shortDisplayName || home.team.displayName}</span>
            ${home.team.logo ? `<img class="logo" src="${home.team.logo}" alt="">` : ""}
          </div>
          <div class="status ${isLive ? "live" : ""}">
            ${isLive ? "● LIVE · " : ""}${status.detail || status.description}
          </div>
        </div>
      `;
    }

    async function loadScores() {
      const content = document.getElementById("content");
      content.innerHTML = `<div class="loading">Loading ${currentSport} scores...</div>`;

      try {
        const leagues = SPORTS[currentSport];
        let html = "";

        for (const league of leagues) {
          const data = await fetchScoreboard(league.path);
          const events = data.events || [];

          if (events.length === 0) continue;

          html += `<div class="league"><h2>${league.name}</h2>`;
          events.forEach(e => { html += renderMatch(e); });
          html += `</div>`;
        }

        content.innerHTML = html || `<div class="loading">No games scheduled right now.</div>`;
      } catch (err) {
        content.innerHTML = `<div class="error">Failed to load scores. Try again later.</div>`;
        console.error(err);
      }
    }

    // Tab switching
    document.getElementById("tabs").addEventListener("click", (e) => {
      if (e.target.classList.contains("tab")) {
        document.querySelectorAll(".tab").forEach(t => t.classList.remove("active"));
        e.target.classList.add("active");
        currentSport = e.target.dataset.sport;
        loadScores();
      }
    });

    // Initial load + auto-refresh
    loadScores();
    refreshInterval = setInterval(loadScores, 60000);
  </script>
